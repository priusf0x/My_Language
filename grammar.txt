G = var_init ";" | func_def 

stmt = if             |
       while          |
       var_init  ";"  |
       return    ";"  |
       '{'{stmt} *'}' |
       expr      ";"  |

if = "if" '(' expr ')' stmt
while = "while" '(' expr ')' stmt
func_def = "function" id '(' {var_init} ')' stmt
var_init = "var" id ("=" expr) | ()
return = "return" expr 
assignment = "=" expr
expr = (term () | assignment) {"+" | "-" term}*
bool = primary "==" | ... | "!=" primary
primary = '(' expr ')'                      |
          (id | id '(' id {',' id}* ')'     |
          number                            |

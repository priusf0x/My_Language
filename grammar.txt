global ::= statement

statement ::= while 
            | if
            | function_definiton
            | function ";"
            | initialization_var ";"
            | assignment 
            | { statement }*
while ::= "while" "(" expression ")" statement
if ::= "if" "(" expression ")" statement
assignment ::= id "=" expression ";"
function_definiton ::= "function" id "(" {initialization_var}* ")" statement
function ::= id "(" {var}* ")" 
initialization_var ::= "var" var ["=" expression]

exrpession ::= term { ("+" | "-") term}*
term ::= bool [("/" | "*") bool]*
bool ::= primary [("<" | ">" | etc) primary]*
primary ::= "(" expression ")" | function | var | const 
var ::= id
const ::= {digits}*

digits ::= 0 | 1 | etc 
id ::= не цифра | все кроме спейсов всяких
